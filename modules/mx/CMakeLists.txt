set(the_description "MX-C500 backend using MACA Runtime and CV-CUDA")
ocv_define_module(mx opencv_core opencv_imgproc OPTIONAL opencv_python_bindings_generator)
find_path(MX_RUNTIME_INCLUDE_DIR mcr/mc_runtime_api.h HINTS ${MACA_PATH} ENV MACA_PATH PATHS /opt/maca-3.8.2 PATH_SUFFIXES include)
find_library(MX_RUNTIME_LIBRARY NAMES mcruntime HINTS ${MACA_PATH} ENV MACA_PATH PATHS /opt/maca-3.8.2 PATH_SUFFIXES lib lib64)
find_library(MX_CUDA_BRIDGE_LIBRARY NAMES runtime_cu HINTS ${MACA_PATH} ENV MACA_PATH PATHS /opt/maca-3.8.2 PATH_SUFFIXES lib lib64)
find_path(MX_CUDA_BRIDGE_INCLUDE_DIR cuda_runtime.h HINTS ${MACA_PATH} ENV MACA_PATH PATHS /opt/maca-3.8.2 PATH_SUFFIXES tools/cu-bridge/include)
find_path(CVCUDA_INCLUDE_DIR cvcuda/OpResize.h HINTS ${CVCUDA_ROOT} ENV CVCUDA_ROOT PATHS /opt/maca-ai/cvcuda0 PATH_SUFFIXES include)
find_library(CVCUDA_LIBRARY NAMES cvcuda HINTS ${CVCUDA_ROOT} ENV CVCUDA_ROOT PATHS /opt/maca-ai/cvcuda0 PATH_SUFFIXES lib lib64 lib/x86_64-linux-gnu)
find_library(NVCV_TYPES_LIBRARY NAMES nvcv_types HINTS ${CVCUDA_ROOT} ENV CVCUDA_ROOT PATHS /opt/maca-ai/cvcuda0 PATH_SUFFIXES lib lib64 lib/x86_64-linux-gnu)
foreach(dep MX_RUNTIME_INCLUDE_DIR MX_RUNTIME_LIBRARY MX_CUDA_BRIDGE_LIBRARY MX_CUDA_BRIDGE_INCLUDE_DIR CVCUDA_INCLUDE_DIR CVCUDA_LIBRARY NVCV_TYPES_LIBRARY)
  if(NOT ${dep})
    message(FATAL_ERROR "MX backend dependency not found: ${dep}")
  endif()
endforeach()
ocv_target_sources(${the_module} PRIVATE src/gpumat.cpp src/resize.cpp src/cvtcolor.cpp)
ocv_target_include_directories(${the_module} PRIVATE ${MX_RUNTIME_INCLUDE_DIR} ${MX_RUNTIME_INCLUDE_DIR}/mcr ${MX_CUDA_BRIDGE_INCLUDE_DIR} ${CVCUDA_INCLUDE_DIR})
ocv_target_link_libraries(${the_module} PRIVATE ${MX_RUNTIME_LIBRARY} ${MX_CUDA_BRIDGE_LIBRARY} ${CVCUDA_LIBRARY} ${NVCV_TYPES_LIBRARY})
